import socket, json

sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
sock.connect(("127.0.0.1", 9090))
sock_file = sock.makefile("r", encoding="utf-8")

# Push High-Rate IMU Frame (10 Hz up to 200 Hz)
frame = {
    "type": "IMU", "timestamp_s": 10.2,
    "ax_mps2": 0.15, "ay_mps2": 0.0, "az_mps2": 9.81,
    "gx_rads": 0.0, "gy_rads": 0.0, "gz_rads": 0.02
}
sock.sendall((json.dumps(frame) + "\n").encode("utf-8"))

# Read back real-time DriftCorrectedState
state_json = sock_file.readline()
